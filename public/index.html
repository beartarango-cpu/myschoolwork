function browse() { let target = document.getElementById('siteUrl').value.trim(); if (!target) return alert('Enter a URL or search query first'); // Check if the input is a plain search query instead of a URL const isUrlPattern = /^(https?:\/\/)?([\da-z\.-]+)\.([a-z\.]{2,6})([\/\w \.-]*)*\/?$/i; const hasSpaces = target.includes(' '); if (!isUrlPattern.test(target) || hasSpaces) { // COOKIE BYPASS: Route search phrases through DuckDuckGo's lightweight engine target = 'https://duckduckgo.com' + encodeURIComponent(target); } else { if (!/^https?:\/\//i.test(target)) { target = 'https://' + target; } } // YouTube Embed optimization filters if (target.includes('youtube.com/watch?v=')) { const urlParams = new URL(target).searchParams; const videoId = urlParams.get('v'); if (videoId) { target = 'https://youtube-nocookie.com' + videoId; } } else if (target.includes('youtu.be/')) { const urlParts = target.split('youtu.be/'); const videoId = urlParts ? urlParts[1].split(/[?#]/)[0] : null; if (videoId) { target = 'https://youtube-nocookie.com' + videoId; } } // Send request to your proxy endpoint window.location.href = window.location.origin + "/proxy?url=" + encodeURIComponent(target); }
